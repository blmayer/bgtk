+static void bgtk_log_timestamp(char *ts, size_t tslen)

+	struct tm tm;

+		snprintf(ts + n, tslen - (size_t)n, ".%03d",

+static int bgtk_mkdir_p(const char *path)
+{
+	struct stat st;
+	if (!path || !path[0])
+		return -1;
+	if (stat(path, &st) == 0)
+		return S_ISDIR(st.st_mode) ? 0 : -1;
+	if (mkdir(path, 0700) < 0 && errno != EEXIST)
+		return -1;
+	return 0;
