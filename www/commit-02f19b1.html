+/* Create every parent component of path (mkdir -p). path is a file path. */
+static void bgtk_mkdir_parents(const char *path)
+{
+	char dir[512];
+	size_t n;
+	char *p;
+
+	if (!path)
+		return;
+	n = strlen(path);
+	if (n == 0 || n >= sizeof(dir))
+		return;
+	memcpy(dir, path, n + 1);
+	p = strrchr(dir, '/');
+	if (!p || p == dir)
+		return;
+	*p = '\0';
+	for (p = dir + 1; *p; p++) {
+		if (*p != '/')
+			continue;
+		*p = '\0';
+		mkdir(dir, 0755);
+		*p = '/';
+	}
+	mkdir(dir, 0755);
+}
